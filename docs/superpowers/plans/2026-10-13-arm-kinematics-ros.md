# Arm Kinematics and ROS 2 Packages Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn an exported robot zip into a ROS 2 Jazzy workspace (description, MoveIt config with TRAC-IK, all-solutions IK package via EAIK) and prove it works in a Docker container.

**Architecture:** A new backend package `backend/app/ros/` parses the URDF into a serial chain (`chain.py`), does numpy FK (`fk.py`), samples self-collisions for the SRDF (`collisions.py`), and writes every ROS file as plain text (`moveit_config.py`, `kinematics_pkg.py`), behind a CLI (`cli.py`). `chain.py` and `fk.py` are stdlib + numpy only and are copied into the generated `<robot>_kinematics` package. Container checks in `docker/jazzy/` build the workspace, launch MoveIt on mock hardware and verify controllers, TF, motion and IK.

**Tech Stack:** Python 3.11 (backend, conda env), numpy, python-fcl, trimesh; ROS 2 Jazzy, MoveIt 2, `trac_ik_kinematics_plugin`, `ros2_control`, EAIK (container only); Docker Desktop on Windows 11.

**Spec:** `docs/superpowers/specs/2026-10-13-arm-kinematics-ros-design.md`

## Global Constraints

- Backend tests run from `backend/` with `OPENBLAS_NUM_THREADS=1 ~/miniforge3/envs/cad-to-urdf/python.exe -m pytest` (Git Bash). Below, `PY` means `~/miniforge3/envs/cad-to-urdf/python.exe`.
- No new backend dependencies. EAIK is installed only in the container (`pip install EAIK` is Linux-only).
- `backend/app/ros/chain.py` and `fk.py`: stdlib + numpy only, relative imports only (`from .chain import ...`); they are copied into generated packages.
- ROS 2 distro: Jazzy. Image: `FROM ros:jazzy`, tag `cad-to-urdf-jazzy`.
- Package names: `<robot>`, `<robot>_moveit_config`, `<robot>_kinematics`. Robot name must match `^[a-z][a-z0-9_]*$`.
- MoveIt IK: `trac_ik_kinematics_plugin/TRAC_IKKinematicsPlugin`, `solve_type: Distance`, `kinematics_solver_timeout: 0.05`; commented pick_ik alternative.
- Placeholder limits: `max_velocity: 1.0`, `max_acceleration: 2.0`, comment `set from datasheet`.
- Collision samples default 10000. IK pose tolerance 1e-6 (m and rad).
- Every generated file in `<robot>_moveit_config` and `<robot>_kinematics` (except the empty `resource/` marker) starts with this note (`#` or XML comment):
  `Generated by cad-to-urdf from <zip name>. Regenerating overwrites this file.` / `Put your own code in <robot>_tasks/.`
- Generated text files are written with `\n` line endings (`newline="\n"`).
- The CLI writes nothing partial: any error leaves the output directory untouched.
- Commits end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```python
"""Small robots exported through the real exporter, for app.ros tests."""
import zipfile
from pathlib import Path

import trimesh

from app.export.urdf_writer import export_package
from app.models import Body, Joint, Link


def _link(id_):
    return Link(id=id_, name=id_, body_ids=[f"{id_}_body"], density_kg_m3=1000.0)


def _body(id_, center):
    box = trimesh.creation.box(extents=(0.1, 0.1, 0.1))
    box.apply_translation(center)
    return Body(id=f"{id_}_body", name=id_, vertices=[tuple(v) for v in box.vertices],
                triangles=[tuple(t) for t in box.faces], faces=[])


def _joint(id_, parent, child, xyz, axis):
    return Joint(id=id_, parent_link_id=parent, child_link_id=child, type="revolute",
                 origin_xyz=xyz, origin_rpy=(0.0, 0.0, 0.0), axis=axis,
                 limit_lower=-1.5, limit_upper=1.5, source="confirmed")


def two_joint_zip(tmp_path: Path, name: str = "bot") -> Path:
    """base -> arm (revolute about z at (0,0,0.2)) -> hand (revolute about y at (0.5,0,0.2)).
    Joints export as arm_joint and hand_joint, limits ±1.5 rad."""
    links = [_link("base"), _link("arm"), _link("hand")]
    joints = [_joint("j1", "base", "arm", (0.0, 0.0, 0.2), (0.0, 0.0, 1.0)),
              _joint("j2", "arm", "hand", (0.5, 0.0, 0.2), (0.0, 1.0, 0.0))]
    bodies = {"base": [_body("base", (0.0, 0.0, 0.0))],
              "arm": [_body("arm", (0.25, 0.0, 0.2))],
              "hand": [_body("hand", (0.7, 0.0, 0.2))]}
    path = tmp_path / f"{name}.zip"
    path.write_bytes(export_package(name, links, joints, bodies))
    return path


def unzip(zip_path: Path, dest: Path) -> Path:
    with zipfile.ZipFile(zip_path) as z:
        z.extractall(dest)
    return Path(dest)
```

```python
"""Serial chain from a URDF: base to tip through the moving joints.

stdlib + numpy only, relative imports only: the generated <robot>_kinematics package ships a
copy of this file."""
import math
import xml.etree.ElementTree as ET
from dataclasses import dataclass

import numpy as np

MOVING = ("revolute", "prismatic", "continuous")


class UnsupportedRobot(ValueError):
    pass


def origin_matrix(el) -> np.ndarray:
    """4x4 transform from a URDF <origin xyz rpy> (rpy = fixed-axis roll, pitch, yaw)."""
    T = np.eye(4)
    if el is None:
        return T
    x, y, z = (float(v) for v in el.get("xyz", "0 0 0").split())
    r, p, w = (float(v) for v in el.get("rpy", "0 0 0").split())
    cr, sr, cp, sp, cw, sw = math.cos(r), math.sin(r), math.cos(p), math.sin(p), math.cos(w), math.sin(w)
    T[:3, :3] = [[cw * cp, cw * sp * sr - sw * cr, cw * sp * cr + sw * sr],
                 [sw * cp, sw * sp * sr + cw * cr, sw * sp * cr - cw * sr],
                 [-sp, cp * sr, cp * cr]]
    T[:3, 3] = (x, y, z)
    return T


@dataclass
class ChainJoint:
    name: str
    type: str
    parent: str
    child: str
    origin: np.ndarray      # previous segment frame -> joint frame at q=0 (fixed joints folded in)
    axis: np.ndarray        # unit vector in the joint frame
    lower: float | None
    upper: float | None


@dataclass
class Chain:
    robot: str
    base: str
    tip: str
    joints: list[ChainJoint]
    links: dict[str, tuple[int, np.ndarray]]  # link -> (segment, pose in segment frame)


def parse_chain(urdf_text: str) -> Chain:
    root = ET.fromstring(urdf_text)
    joints = root.findall("joint")
    children: dict[str, list] = {}
    has_parent = set()
    for j in joints:
        children.setdefault(j.find("parent").get("link"), []).append(j)
        has_parent.add(j.find("child").get("link"))
    roots = [l.get("name") for l in root.findall("link") if l.get("name") not in has_parent]
    if len(roots) != 1:
        raise UnsupportedRobot(f"URDF has {len(roots)} root links; expected one.")

    def moving_below(link: str) -> bool:
        return any(j.get("type") != "fixed" or moving_below(j.find("child").get("link"))
                   for j in children.get(link, []))

    wheels = [j for j in joints
              if j.get("type") == "continuous" and not moving_below(j.find("child").get("link"))]
    if len(wheels) >= 2:
        raise UnsupportedRobot(
            f"Joint '{wheels[0].get('name')}' looks like a wheel; mobile robots are not supported yet.")
    for j in joints:
        if j.get("type") not in MOVING + ("fixed",):
            raise UnsupportedRobot(
                f"Joint '{j.get('name')}' is {j.get('type')}; only revolute, prismatic, "
                "continuous and fixed joints are supported.")
    leaves = [j for j in joints
              if j.get("type") != "fixed" and not moving_below(j.find("child").get("link"))]
    if len(leaves) > 1:
        raise UnsupportedRobot(
            f"Robot has {len(leaves)} moving branches; only single-arm robots are supported yet.")
    if not leaves:
        raise UnsupportedRobot("Robot has no moving joints.")

    chain_joints: list[ChainJoint] = []
    link_poses: dict[str, tuple[int, np.ndarray]] = {}

    def walk(link: str, seg: int, T: np.ndarray) -> None:
        link_poses[link] = (seg, T)
        for j in children.get(link, []):
            child = j.find("child").get("link")
            O = T @ origin_matrix(j.find("origin"))
            if j.get("type") == "fixed":
                walk(child, seg, O)
                continue
            lower = upper = None
            if j.get("type") != "continuous":
                lim = j.find("limit")
                if lim is None or lim.get("lower") is None or lim.get("upper") is None:
                    raise UnsupportedRobot(
                        f"Joint '{j.get('name')}' has no limits; set them in the app and export again.")
                lower, upper = float(lim.get("lower")), float(lim.get("upper"))
            ax_el = j.find("axis")
            axis = np.array([float(v) for v in (ax_el.get("xyz") if ax_el is not None else "1 0 0").split()])
            chain_joints.append(ChainJoint(j.get("name"), j.get("type"), link, child, O,
                                           axis / np.linalg.norm(axis), lower, upper))
            walk(child, len(chain_joints), np.eye(4))

    walk(roots[0], 0, np.eye(4))
    return Chain(root.get("name"), roots[0], chain_joints[-1].child, chain_joints, link_poses)


def home(chain: Chain) -> list[float]:
    """Every joint at 0, or at its nearest limit when 0 is out of range."""
    return [0.0 if j.lower is None else min(max(0.0, j.lower), j.upper) for j in chain.joints]
```

```python
"""Forward kinematics for a Chain. numpy only: the generated <robot>_kinematics package ships a copy."""
import math

import numpy as np

from .chain import Chain, ChainJoint


def _motion(j: ChainJoint, q: float) -> np.ndarray:
    T = np.eye(4)
    if j.type == "prismatic":
        T[:3, 3] = j.axis * q
        return T
    x, y, z = j.axis
    c, s = math.cos(q), math.sin(q)
    C = 1 - c
    T[:3, :3] = [[c + x * x * C, x * y * C - z * s, x * z * C + y * s],
                 [y * x * C + z * s, c + y * y * C, y * z * C - x * s],
                 [z * x * C - y * s, z * y * C + x * s, c + z * z * C]]
    return T


def segment_poses(chain: Chain, q) -> list[np.ndarray]:
    """Base-frame pose of each segment: [base, after joint 1, ..., after joint n]."""
    T = np.eye(4)
    out = [T]
    for j, qi in zip(chain.joints, q, strict=True):
        T = T @ j.origin @ _motion(j, float(qi))
        out.append(T)
    return out


def fk(chain: Chain, q) -> np.ndarray:
    return segment_poses(chain, q)[-1]


def link_poses(chain: Chain, q) -> dict[str, np.ndarray]:
    seg = segment_poses(chain, q)
    return {name: seg[i] @ T for name, (i, T) in chain.links.items()}
```

```python
"""Which link pairs MoveIt may skip when checking self-collision (the SRDF's disable_collisions).

Same idea as MoveIt Setup Assistant: pairs joined by a joint, pairs touching at home, and pairs
that never touch in many random poses are skipped."""
import xml.etree.ElementTree as ET
from itertools import combinations
from pathlib import Path

import fcl
import numpy as np
import trimesh

from app.validate.sweep import collides, fcl_object

from .chain import Chain, home, origin_matrix
from .fk import link_poses


def link_meshes(urdf_text: str, package_root: Path) -> dict[str, tuple[np.ndarray, np.ndarray]]:
    """Collision mesh of each link in its own frame. package://<pkg>/x resolves to package_root/x."""
    out = {}
    for link in ET.fromstring(urdf_text).findall("link"):
        parts = []
        for col in link.findall("collision"):
            mesh = col.find("geometry/mesh")
            if mesh is None:
                continue
            rel = mesh.get("filename").split("://", 1)[-1].split("/", 1)[1]
            m = trimesh.load(Path(package_root) / rel, file_type="stl", force="mesh")
            m.apply_transform(origin_matrix(col.find("origin")))
            parts.append(m)
        if parts:
            m = trimesh.util.concatenate(parts)
            out[link.get("name")] = (np.asarray(m.vertices), np.asarray(m.faces))
    return out


def _random_q(chain: Chain, rng) -> list[float]:
    return [rng.uniform(j.lower, j.upper) if j.lower is not None else rng.uniform(-np.pi, np.pi)
            for j in chain.joints]


def disabled_pairs(chain: Chain, meshes, samples: int = 10000, seed: int = 0) -> dict[tuple[str, str], str]:
    objs = {n: fcl_object(v, f) for n, (v, f) in meshes.items()}
    names = sorted(objs)

    def place(q) -> None:
        for n, T in link_poses(chain, q).items():
            if n in objs:
                objs[n].setTransform(fcl.Transform(T[:3, :3], T[:3, 3]))

    out, open_pairs = {}, []
    for a, b in combinations(names, 2):
        if abs(chain.links[a][0] - chain.links[b][0]) <= 1:
            out[(a, b)] = "Adjacent"
        else:
            open_pairs.append((a, b))
    place(home(chain))
    for p in open_pairs:
        if collides(objs[p[0]], objs[p[1]]):
            out[p] = "Default"
    open_pairs = [p for p in open_pairs if p not in out]
    rng = np.random.default_rng(seed)
    for _ in range(samples):
        if not open_pairs:
            break
        place(_random_q(chain, rng))
        open_pairs = [p for p in open_pairs if not collides(objs[p[0]], objs[p[1]])]
    for p in open_pairs:
        out[p] = "Never"
    return out
```

```python
        "config/ros2_controllers.yaml": _controllers_yaml(names, h),
        "config/moveit_controllers.yaml": _moveit_controllers_yaml(names, h),
        "config/ompl_planning.yaml": _ompl_yaml(h),
        "config/initial_positions.yaml": h + f"initial_positions:\n{initial}\n",
        f"config/{robot}.ros2_control.xacro": _ros2_control_xacro(chain, robot, x),
        f"config/{robot}.urdf.xacro": _urdf_xacro(robot, x),

def generate(robot: str, description_dir: Path, out_src: Path, source: str, samples: int = 10000) -> None:
    """Write <robot>, <robot>_moveit_config (and <robot>_kinematics) into out_src.
    Everything is built in a temp folder first: an error leaves out_src untouched."""
    if not ROS_NAME.match(robot):
        raise ValueError(f"Robot name '{robot}' is not a valid ROS package name; use lowercase "
                         "letters, digits and underscores, starting with a letter.")
    description_dir = Path(description_dir)
    urdf_path = description_dir / "urdf" / f"{robot}.urdf"
    if not urdf_path.exists():
        raise ValueError("Not a cad-to-urdf export.")
    urdf = urdf_path.read_text(encoding="utf-8")
    chain = parse_chain(urdf)
    pairs = disabled_pairs(chain, link_meshes(urdf, description_dir), samples=samples)

    tmp = Path(tempfile.mkdtemp(prefix="cad-to-urdf-"))
    try:
        shutil.copytree(description_dir, tmp / robot)
        write_files(tmp / robot, {"package.xml": package_xml(robot), "CMakeLists.txt": cmakelists(robot)})
        write_files(tmp / f"{robot}_moveit_config", moveit_files(chain, robot, source, pairs))
        out_src = Path(out_src)
        out_src.mkdir(parents=True, exist_ok=True)
        for pkg in sorted(p.name for p in tmp.iterdir()):
            shutil.rmtree(out_src / pkg, ignore_errors=True)
            shutil.move(str(tmp / pkg), str(out_src / pkg))
    finally:
        shutil.rmtree(tmp, ignore_errors=True)
```

```python
"""Every exact IK solution for a pose (EAIK closed form), each checked against this package's FK."""
import math
import os

import numpy as np

from ._chain import parse_chain
from ._fk import fk as _fk

ROBOT = "__ROBOT__"
TOL = 1e-6  # m and rad: every returned solution reproduces the pose at least this closely


def _pose_error(a, b) -> float:
    dp = float(np.linalg.norm(a[:3, 3] - b[:3, 3]))
    c = (np.trace(a[:3, :3].T @ b[:3, :3]) - 1) / 2
    return max(dp, math.acos(max(-1.0, min(1.0, c))))


def _within_limits(q, chain):
    """Wrap revolute angles into the joint's range; None when any joint can't be."""
    # ponytail: one representative per joint; ranges wider than 2π hold more, add them if a robot needs it
    out = []
    for qi, j in zip(q, chain.joints):
        if j.type == "prismatic":
            cands = [float(qi)]
        else:
            r = math.remainder(float(qi), 2 * math.pi)
            cands = [r, r - 2 * math.pi, r + 2 * math.pi]
        if j.lower is not None:
            cands = [c for c in cands if j.lower - 1e-9 <= c <= j.upper + 1e-9]
        if not cands:
            return None
        out.append(cands[0])
    return np.array(out)


class Kinematics:
    def __init__(self, urdf_path: str):
        with open(urdf_path, encoding="utf-8") as f:
            self.chain = parse_chain(f.read())
        self._bot, self._why = None, ""
        try:
            from eaik.IK_URDF import UrdfRobot
            bot = UrdfRobot(urdf_path)
            if hasattr(bot, "hasKnownDecomposition") and not bot.hasKnownDecomposition():
                self._why = "EAIK knows no decomposition for this geometry"
            else:
                z = np.zeros(len(self.chain.joints))
                # EAIK's end frame may differ from the URDF tip by a fixed offset
                self._tip = np.linalg.inv(np.asarray(bot.fwdKin(z), float)) @ _fk(self.chain, z)
                self._bot = bot
        except Exception as exc:  # EAIK refuses geometries it cannot decompose
            self._why = f"{type(exc).__name__}: {exc}"

    def fk(self, q) -> np.ndarray:
        return _fk(self.chain, q)

    def solve_all(self, pose, current=None) -> list:
        """All joint solutions within limits for a 4x4 base-frame pose, nearest to `current` first."""
        if self._bot is None:
            raise NotImplementedError(f"no closed-form solver for this geometry ({self._why})")
        pose = np.asarray(pose, float)
        out = []
        for q in self._bot.IK(pose @ np.linalg.inv(self._tip)).Q:
            q = _within_limits(q, self.chain)
            if (q is not None and _pose_error(self.fk(q), pose) <= TOL
                    and not any(np.allclose(q, o, atol=1e-9) for o in out)):
                out.append(q)
        ref = np.zeros(len(self.chain.joints)) if current is None else np.asarray(current, float)
        return sorted(out, key=lambda q: float(np.linalg.norm(q - ref)))


_default = None


def load() -> Kinematics:
    """The installed robot's kinematics (needs a sourced ROS 2 workspace)."""
    global _default
    if _default is None:
        from ament_index_python.packages import get_package_share_directory
        _default = Kinematics(os.path.join(get_package_share_directory(ROBOT), "urdf", f"{ROBOT}.urdf"))
    return _default


def fk(q):
    return load().fk(q)


def solve_all(pose, current=None):
    return load().solve_all(pose, current)
```

```python
"""Generate a ROS 2 Jazzy workspace from a cad-to-urdf export.

Usage: python -m app.ros.cli <export.zip> -o <workspace dir> [--samples N]"""
import argparse
import re
import sys
import tempfile
import zipfile
from pathlib import Path

from .moveit_config import generate


def _robot_name(z: zipfile.ZipFile) -> str:
    """The export holds <robot>/urdf/<robot>.urdf."""
    found = [n.split("/")[0] for n in z.namelist() if re.fullmatch(r"(?P<r>[^/]+)/urdf/(?P=r)\.urdf", n)]
    if len(found) != 1:
        raise ValueError("Not a cad-to-urdf export.")
    return found[0]


def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(prog="python -m app.ros.cli")
    p.add_argument("zip")
    p.add_argument("-o", "--out", required=True, help="workspace dir; packages go in <out>/src")
    p.add_argument("--samples", type=int, default=10000, help="random poses for the collision matrix")
    a = p.parse_args(argv)
    src = Path(a.out) / "src"
    try:
        with zipfile.ZipFile(a.zip) as z, tempfile.TemporaryDirectory() as tmp:
            robot = _robot_name(z)
            z.extractall(tmp)
            generate(robot, Path(tmp) / robot, src, source=Path(a.zip).name, samples=a.samples)
    except zipfile.BadZipFile:
        print("error: Not a cad-to-urdf export.", file=sys.stderr)
        return 1
    except (ValueError, OSError) as exc:
        print(f"error: {exc}", file=sys.stderr)
        return 1
    print(f"wrote {src}: {robot}, {robot}_moveit_config, {robot}_kinematics")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Helpers shared by the container checks."""
import importlib
import math
import time

import numpy as np
import rclpy


def load_kinematics(robot: str):
    return importlib.import_module(f"{robot}_kinematics").load()


def quat(R) -> tuple[float, float, float, float]:
    """(x, y, z, w) from a rotation matrix."""
    w = math.sqrt(max(0.0, 1 + R[0, 0] + R[1, 1] + R[2, 2])) / 2
    x = math.copysign(math.sqrt(max(0.0, 1 + R[0, 0] - R[1, 1] - R[2, 2])) / 2, R[2, 1] - R[1, 2])
    y = math.copysign(math.sqrt(max(0.0, 1 - R[0, 0] + R[1, 1] - R[2, 2])) / 2, R[0, 2] - R[2, 0])
    z = math.copysign(math.sqrt(max(0.0, 1 - R[0, 0] - R[1, 1] + R[2, 2])) / 2, R[1, 0] - R[0, 1])
    return x, y, z, w


def set_pose(msg, T) -> None:
    """Fill a geometry_msgs/Pose from a 4x4."""
    msg.position.x, msg.position.y, msg.position.z = (float(v) for v in T[:3, 3])
    msg.orientation.x, msg.orientation.y, msg.orientation.z, msg.orientation.w = quat(T[:3, :3])


def pose_error(a, b) -> float:
    dp = float(np.linalg.norm(a[:3, 3] - b[:3, 3]))
    c = (np.trace(a[:3, :3].T @ b[:3, :3]) - 1) / 2
    return max(dp, math.acos(max(-1.0, min(1.0, c))))


def same_angles(a, b, tol) -> bool:
    d = np.remainder(np.asarray(a) - np.asarray(b) + math.pi, 2 * math.pi) - math.pi
    return bool(np.all(np.abs(d) <= tol))


def wait(node, future, timeout: float):
    end = time.time() + timeout
    while not future.done() and time.time() < end:
        rclpy.spin_once(node, timeout_sec=0.1)
    if not future.done():
        raise TimeoutError("no reply")
    return future.result()


def report(name: str, ok: bool, detail: str) -> bool:
    print(f"{'PASS' if ok else 'FAIL'} {name}: {detail}", flush=True)
    return ok
```

```python
def main(robot: str) -> int:

    chain = kin.chain
    rclpy.init()
    node = Node("cad_to_urdf_check")
    ok = True

    cm = node.create_client(ListControllers, "/controller_manager/list_controllers")
    states, end = {}, time.time() + 60
    while time.time() < end:
        if cm.wait_for_service(timeout_sec=1.0):
            states = {c.name: c.state for c in wait(node, cm.call_async(ListControllers.Request()), 5).controller}
            if all(states.get(n) == "active" for n in ("arm_controller", "joint_state_broadcaster")):
                break

                 all(states.get(n) == "active" for n in ("arm_controller", "joint_state_broadcaster")),
                 str(states))

    buf = Buffer()
    TransformListener(buf, node)
    found, end = False, time.time() + 15
    while not found and time.time() < end:
        rclpy.spin_once(node, timeout_sec=0.2)
        found = buf.can_transform(chain.base, chain.tip, Time())
    ok &= report("TF base to tip", found, f"{chain.base} -> {chain.tip}")

    q = []
    for i, j in enumerate(chain.joints):
        v = 0.3 if i % 2 == 0 else -0.3
        q.append(v if j.lower is None else min(max(v, j.lower), j.upper))
    target = kin.fk(q)

    goal = MoveGroup.Goal()
    req = goal.request
    req.group_name = "arm"
    req.num_planning_attempts = 5
    req.allowed_planning_time = 10.0
    req.max_velocity_scaling_factor = 0.5
    req.max_acceleration_scaling_factor = 0.5
    pc = PositionConstraint()
    pc.header.frame_id, pc.link_name, pc.weight = chain.base, chain.tip, 1.0
    pc.constraint_region.primitives.append(SolidPrimitive(type=SolidPrimitive.SPHERE, dimensions=[1e-3]))
    centre = Pose()
    set_pose(centre, target)
    centre.orientation.x = centre.orientation.y = centre.orientation.z = 0.0
    centre.orientation.w = 1.0
    pc.constraint_region.primitive_poses.append(centre)
    oc = OrientationConstraint()
    oc.header.frame_id, oc.link_name, oc.weight = chain.base, chain.tip, 1.0
    oc.orientation.x, oc.orientation.y, oc.orientation.z, oc.orientation.w = quat(target[:3, :3])
    oc.absolute_x_axis_tolerance = oc.absolute_y_axis_tolerance = oc.absolute_z_axis_tolerance = 1e-3
    req.goal_constraints.append(Constraints(position_constraints=[pc], orientation_constraints=[oc]))
    goal.planning_options.plan_only = False

    ac = ActionClient(node, MoveGroup, "/move_action")
    code = None
    if ac.wait_for_server(timeout_sec=30):
        handle = wait(node, ac.send_goal_async(goal), 30)
        code = wait(node, handle.get_result_async(), 120).result.error_code.val

    latest = {}

    q_now = [latest.get(j.name, math.nan) for j in chain.joints]
    err = pose_error(kin.fk(q_now), target) if not np.isnan(q_now).any() else math.inf
    ok &= report("joint_states at target", err < 2e-3, f"pose error {err:.2e}")

    rclpy.shutdown()
    return 0 if ok else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv[1]))
```

```python
"""Container check: EAIK returns every IK solution and each reproduces its pose; TRAC-IK agrees.
Usage: python3 check_ik.py <robot>   (demo.launch.py must be running for /compute_ik)"""
import importlib
import math
import sys

import numpy as np
import rclpy
from moveit_msgs.srv import GetPositionIK
from rclpy.node import Node

from common import load_kinematics, pose_error, report, same_angles, set_pose, wait

POSES = 200


def main(robot: str) -> int:
    kin = load_kinematics(robot)
    chain = kin.chain
    home = importlib.import_module(f"{robot}_kinematics._chain").home(chain)
    names = [j.name for j in chain.joints]
    rng = np.random.default_rng(1)
    rclpy.init()
    node = Node("cad_to_urdf_ik_check")
    ik = node.create_client(GetPositionIK, "/compute_ik")
    if not ik.wait_for_service(timeout_sec=60):
        report("TRAC-IK service", False, "/compute_ik not available")
        return 1

    counts, worst, missing, trac_ok, trac_match = [], 0.0, 0, 0, 0
    for _ in range(POSES):
        q = np.array([rng.uniform(j.lower, j.upper) if j.lower is not None else rng.uniform(-math.pi, math.pi)
                      for j in chain.joints])
        pose = kin.fk(q)
        sols = kin.solve_all(pose)
        counts.append(len(sols))
        worst = max([worst] + [pose_error(kin.fk(s), pose) for s in sols])
        if not any(same_angles(s, q, 1e-4) for s in sols):
            missing += 1

        req = GetPositionIK.Request()
        r = req.ik_request
        r.group_name, r.ik_link_name, r.avoid_collisions = "arm", chain.tip, False
        r.timeout.nanosec = 50_000_000
        r.pose_stamped.header.frame_id = chain.base
        set_pose(r.pose_stamped.pose, pose)
        r.robot_state.joint_state.name = names
        r.robot_state.joint_state.position = [float(v) for v in home]
        res = wait(node, ik.call_async(req), 5)
        if res.error_code.val == 1:
            got = dict(zip(res.solution.joint_state.name, res.solution.joint_state.position))
            qt = np.array([got[n] for n in names])
            if pose_error(kin.fk(qt), pose) < 1e-4:
                trac_ok += 1
                trac_match += any(same_angles(qt, s, 1e-3) for s in sols)

    ok = report("EAIK finds the sampled pose's joints", missing == 0, f"missing {missing}/{POSES}")
    ok &= report("EAIK solutions reproduce their pose", worst <= 1e-6 and min(counts) >= 1,
                 f"worst error {worst:.1e}; solutions per pose min {min(counts)} max {max(counts)} "
                 f"mean {np.mean(counts):.1f}")
    ok &= report("TRAC-IK success rate", trac_ok >= 0.95 * POSES, f"{trac_ok}/{POSES}")
    ok &= report("TRAC-IK answer is one of EAIK's", trac_match == trac_ok, f"{trac_match}/{trac_ok}")
    rclpy.shutdown()
    return 0 if ok else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv[1]))
```
